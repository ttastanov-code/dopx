# Пример реализации PlayerEvaluation:
class PlayerEvaluation(BaseModel):
    user = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.CASCADE)
    match = models.ForeignKey(Match, on_delete=models.CASCADE)
    player = models.ForeignKey(Player, on_delete=models.CASCADE)
    
    contribution = models.IntegerField(validators=[MinValueValidator(1), MaxValueValidator(10)])
    risk = models.IntegerField(validators=[MinValueValidator(1), MaxValueValidator(10)])
    potential = models.IntegerField(validators=[MinValueValidator(1), MaxValueValidator(10)])
    
    class Meta:
        constraints = [
            models.UniqueConstraint(fields=['user', 'match', 'player'], name='unique_player_eval')
        ]

def calculate_user_weight(user: User, context_eval: ContextEvaluation) -> float:
    weight = 1.0
    if context_eval.watched_type == 'full': weight += 0.2
    if user.trust_score > 1.2: weight += 0.2
    # TODO: детекция "фанатской системности"
    return weight
